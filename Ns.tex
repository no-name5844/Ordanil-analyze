\section{Ns(Nonlinear successo, by 1:送到本到(UTF-8) 2:nnc)}
\subsection{定义}
\subsubsection{辅助函数}
\[
f(\alpha)=\begin{cases}
  1 & \alpha=0\\
  f(\beta)*2 & \alpha=\beta+1\\
  f(expand(\alpha,2)) & \lnot\exists\beta(\alpha=\beta+1)
\end{cases}\qquad(\alpha\in On)
\]

\subsubsection{定义}
对于 $\forall A(\|A\|\ge\aleph_0)$：
\begin{enumerate}
  \item $\alpha\,th\,A\in A$
  \item $1\,st,A=\min A$
  \item $\alpha\,th\,A=\min\{x\mid x>\max\{\beta\,th\,A\mid\beta<\alpha\}\land x\in A\}$
\end{enumerate}
对于 $\mathbb{NS}$：
\begin{enumerate}
  \item $1\,th,\mathbb{NS}=0$
  \item $\alpha+1\,th\,\mathbb{NS}=\alpha\,th\,\mathbb{NS}+\frac{1}{f(\alpha)}$
  \item $\alpha\,th\,\mathbb{NS}=\lim_{n\to\omega}expand(\alpha,n)\,th\,\mathbb{NS},\ \lnot\exists\beta(\alpha=\beta+1)$
\end{enumerate}
% expand(\alpha,n) 即 \alpha[n]
% 主体
对于 $\psi(x)$：
\begin{enumerate}
  \item $\psi(x)\,th\,\mathbb{NS}=x$
\end{enumerate}
