\section{$\varepsilon_p$SS}
\subsection{定义}
$\varepsilon_p$SS 是序数记号中序列记号中的自然数序列中的阶差型记号。极限表达式为 $1,n$
该记号在标准型下满足字典序比较

对于 $A=(a_1,a_2,\dots,a_n)$ 一定满足：
\begin{enumerate}
  \item $a_i\in\mathbb{N}$
  \item $a_1=1$
\end{enumerate}
对于 $A=(a_1,a_2,\dots,a_n)$ 中的元素 $a_i$ 有：
\begin{enumerate}
  \item 列标为 $i$
  \item $i\,th\,A=a_i$
\end{enumerate}

\subsection{展开}
\subsubsection{辅助函数}
\begin{enumerate}
  \item $(a_1,a_2,\dots,a_n)\oplus(b_1,b_2,\dots,b_n)=(a_1,a_2,\dots,a_n,b_1,b_2,\dots,b_n)$
  \item $A\bigoplus^n_{i=a}f(i)=(A\bigoplus^{n-1}_{i=a}f(i))\oplus f(n)$
\end{enumerate}

\subsubsection{expand 函数}
对于 $A=()$ 中：
\begin{gather*}
expand(A,m)=()\\
expandLen(A,m)=()
\end{gather*}
对于 $A=(a_1,a_2,\dots,a_n,1)$ 中：
\begin{gather*}
expand(A,m)=(a_1,a_2,\dots,a_n)\\
expandLen(A,m)=(a_1,a_2,\dots,a_n)
\end{gather*}
对于 $A=(a_1,a_2,\dots,a_n)$ 以上不满足时：
\begin{itemize}
  \item 设 $br=\max\{n-i\mid a_{n-i}<a_n\}$，即从右到左的寻找比 $a_n$ 小的第一个元素的列标。$L=n-br$
  \begin{itemize}
    \item 若 $a_n=a_k+1$
    \begin{gather*}
    G=(a_1,a_2,\dots,a_{br-1})\\
    B=(a_{br},\dots,a_{n-1})\\
    expand(A,m)=G(\bigoplus^m_{i=0}B)
    \end{gather*}
    \item 若 $a_n-a_k=q\le p,\ q>1$
    \begin{gather*}
    G=(a_1,a_2,\dots,a_{n-1})\\
    B_i=(a_n-1+p*i)\\
    expand(A,m)=G(\bigoplus^m_{i=1}B_{i-1})
    \end{gather*}
    \item 若 $a_n-a_k>p$
    \begin{gather*}
    G=(a_1,a_2,\dots,a_{n-1})\\
    B_i=(a_n-1+(q-1)*i)\\
    expand(A,m)=G(\bigoplus^m_{i=1}B_{i-1})
    \end{gather*}
  \end{itemize}
\end{itemize}
