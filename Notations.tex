\section{Ordinal Notations}
\subsection{定义}
\begin{align*}
\mathcal{O}&=(Exp,Zero,Succ,Limit,Prec,FS)\\
Zero&\in Exp\\
Succ&\subset Exp\\
Limit&\subset Exp\\
Prec&: Succ\to Exp\\
FS&: Limit\times\omega\to Exp\\
Exp&=Zero\cup Succ\cup Limit\\
Limit&=\{\alpha\in Exp\mid\alpha\not\in Zero\cup Succ\}\\
FS(\alpha,n)&\text{ 记作 }\alpha[n],\ \langle\alpha[i]\rangle_{i\in\omega}\text{ 称基本列}\\
\mathcal{F}&: Exp\leftrightarrow A\subset On\\
\mathcal{F}(Zero)&=0\\
\mathcal{F}(\alpha)&=\mathcal{F}(Prec(\alpha))+1,\quad\alpha\in Succ\\
\mathcal{F}(\alpha)&=\sup\{\mathcal{F}(FS(\alpha,i))\mid i\in\omega\}\quad\alpha\in Limit\\
&\qquad\land(\forall n\in\omega\ \forall m\in n)(\mathcal{F}(FS(\alpha,n))>\mathcal{F}(FS(\alpha,m)))
\end{align*}

\section{Large Number Notations}
\subsection{定义}
\begin{align*}
\mathcal{N}&=(Exp,FS)\\
\mathbb{N}&\subset Exp\\
FS&: Exp\to Exp\\
\mathcal{F}&: Exp\to\mathbb{N}\\
\mathcal{F}(\alpha)&=\alpha,\quad\alpha\in\mathbb{N}\\
\mathcal{F}(\alpha)&=\mathcal{F}(FS(\alpha)),\quad\alpha\in Exp-\mathbb{N}
\end{align*}
